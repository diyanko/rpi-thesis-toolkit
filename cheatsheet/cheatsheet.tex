% ============================ 1. FILES
\begin{fg}{1}{Your files}{what each one is for}
\begin{tabular}{@{}>{\ttfamily\color{inkblue}}l@{\hspace{6pt}}l@{}}
\row{main.tex}{The file you compile. Lists the parts in order.}
\row{metadata.tex}{Title, name, degree, department, committee.}
\row{frontmatter/}{Abstract and acknowledgment.}
\row{chapters/}{One \file{.tex} file per chapter.}
\row{appendices/}{Extra material, after the references.}
\row{references.bib}{Your sources, one entry each.}
\row{figures/}{Images: PNG, JPEG or PDF.}
\row{preamble.tex}{Citation style and extra packages.}
\row{rpi-thesis.cls}{The formatting. Leave it alone.}
\row{assets/}{The RPI seal: download it (see README).}
\end{tabular}
\begin{snip}
% metadata.tex: change only what is inside { }
\thesissetup{
  title  = {Your thesis title},
  author = {Your full name},
  degree = doctoral, % or masters, engineering-doctorate
  graduation-month = May, % May, August, December
  graduation-year  = {2027},
}
\committeemember[Chair]{Your committee chair}
\committeemember{Committee member two}
\end{snip}
\note{List every approved member, with \code{[Chair]} on one line. Front-matter
pages begin \cs{frontchapter}\code{\{ABSTRACT\}}.}
\note{New chapter? Copy one, retitle it, add its \cs{include} to \file{main.tex}.}
\end{fg}

% ============================ 2. WRITE
\begin{fg}{2}{Write}{plain text, a few commands}
\begin{snip}
\section{Research question}
\subsection{Context}
Ordinary sentences make a paragraph.

A blank line starts the next paragraph.
\emph{emphasis}, \textbf{bold}, \enquote{a short quote}
\begin{quote}
  A long quotation, set apart.
\end{quote}
\begin{itemize}   % enumerate numbers: 1. 2. 3.
  \item First point
  \item Second point
\end{itemize}
\end{snip}
\begin{tabular}{@{}l@{\hspace{6pt}}*{7}{c@{\hspace{8pt}}}@{}}
Type &\code{\textbackslash\%}&\code{\textbackslash\&}&\code{\textbackslash\_}
  &\code{\textbackslash\$}&\code{\textbackslash\#}&\code{\textbackslash\{}
  &\code{\textbackslash\}}\\
to get &\%&\&&\_&\$&\#&\{&\}
\end{tabular}
\note{\code{\%} starts a private comment. Don't type heading numbers or add
space with \code{\textbackslash\textbackslash}. Use one capitalization style for headings and captions.}
\end{fg}
\gap
% ============================ 3. REFER
\begin{fg}{3}{Refer}{label it, then point at it}
\begin{snip}
\label{fig:growth}  % after \caption or \chapter
As Figure~\ref{fig:growth} shows, the trend
rises (Table~\ref{tab:x}, Equation~\ref{eq:x}).
\end{snip}
\note{\code{\textasciitilde} keeps word and number together. Never type
numbers by hand. Prefixes: \code{chap: sec: fig: tab: eq:}}
\end{fg}
\end{minipage}%

% ============================ 4. CITE
\begin{fg}{4}{Cite}{one entry, one key}
\begin{snip}
An earlier account explains this~\cite{lamport1994}.
\fullcite{lamport1994}  % the whole entry, in text
\end{snip}
\begin{bibsnip}
@book{lamport1994,
  author    = {Leslie Lamport},
  title     = {{LaTeX}: A Document Preparation System},
  publisher = {Addison-Wesley},
  date      = {1994}
}
@online{example-page,
  author  = {{Name of the Organization}},
  title   = {Title of the Web Page},
  url     = {https://example.org/page},
  urldate = {2026-10-14}
}
\end{bibsnip}
\note{OGE wants an \textbf{access date for every URL} (not for DOIs):
\code{urldate} is the day \emph{you} read it. Check that it prints.}
\note{In \file{references.bib}, the key follows \code{@book\{}. Keys must be
unique; only cited sources are listed. Braces keep capitals: \code{\{RPI\}}.}
\note{Agree on a style with your advisor and set it in \file{preamble.tex}.
The sample's numeric style is not an official one.}
\end{fg}
\gap
% ============================ 5. FIGURE
\begin{fg}{5}{Figure}{caption below}
\begin{snip}
\begin{figure}[htbp]
  \centering
  \includegraphics[width=\linewidth]{figures/x}
  \caption{One full caption, in your heading style.}
  \label{fig:x}
\end{figure}
\end{snip}
\note{\cs{label} goes \emph{after} \cs{caption}. Never use
\cs{caption}\code{[short]\{long\}}: the list of figures must match the caption.}
\note{Someone else's figure, or one from your paper? Its caption needs the full
citation and a permission or license statement.}
\end{fg}
\gap
% ============================ 6. TABLE
\begin{fg}{6}{Table}{caption above}
\begin{snip}
\begin{table}[htbp]
  \centering
  \caption{Caption goes above the table.}
  \label{tab:x}
  \begin{tabular}{lr}  % l, c, r: column alignment
    \toprule
    Group & Value \\     % & next column, \\ end row
    \midrule
    A     & 12 \\
    \bottomrule
  \end{tabular}
\end{table}
\end{snip}
\note{The class loads \file{booktabs}, which gives \cs{toprule},
\cs{midrule} and \cs{bottomrule}.}
\end{fg}
\end{minipage}%

% ============================ 7. EQUATION
\begin{fg}{7}{Equation}{every display is numbered}
\begin{snip}
Inline maths uses dollar signs: $x = 2$.
\begin{equation}
  y = mx + b.
  \label{eq:line}
\end{equation}
\begin{equation}
  \begin{split}
    (a + b)^2 &= (a + b)(a + b) \\
              &= a^2 + 2ab + b^2.
  \end{split}
  \label{eq:square}
\end{equation}
\end{snip}
\note{Not \code{equation*} or \code{\textbackslash[\,\textbackslash]}: only
displays inside a lemma, theorem, proof or algorithm may go unnumbered.}
\note{Number dropped below? The line is too long; break it with
\code{split} (\code{\&} aligns, \code{\textbackslash\textbackslash} ends a line).}
\end{fg}
\gap
% ============================ 8. PUBLISHED CHAPTER
\begin{fg}{8}{Published chapter}{your own article}
\begin{snip}
\chapter{RESEARCH AND FINDINGS}
\label{chap:research}
\chapterattribution{[Exact sentence from the
  current OGE manual] \fullcite{my-article}.}
\end{snip}
\note{\textbf{Copy the exact OGE wording} for your article's status
(published, accepted or submitted). Co-authored? Also list it on the
Declaration of Co-Contribution (not needed if you are the sole author).}
\end{fg}
